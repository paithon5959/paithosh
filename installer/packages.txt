# Paithosh Shell — Danh sach Package theo tung Distro
# Nguon tham chieu: installer/lib/audit.sh
#
# Luu y ve KDE Connect (bat buoc, chi su dung CLI `kdeconnect-cli`, khong can mo app GUI):
#   - Paithosh chi goi `kdeconnect-cli` (qua KdeConnectService.qml).
#   - Tren Fedora: goi `kdeconnectd` chi co daemon, KHONG co `kdeconnect-cli`.
#     Goi nho nhat chua `kdeconnect-cli` la `kde-connect` (khong can cai them plasma/kde-desktop).
#   - Tren Arch / Debian / Ubuntu: goi duy nhat la `kdeconnect` (da bao gom `kdeconnect-cli`).

# [system] Am thanh, den nen, mang, pin, quyen polkit, luu tru, VPN
wireplumber
pipewire              # Cung cap `pw-play` cho am thanh UI
libpulse              # Cung cap `pactl` cho mix audio khi quay man hinh
brightnessctl
networkmanager
upower
power-profiles-daemon
polkit
hyprpolkitagent
xdg-utils
unzip
pciutils
gsettings-desktop-schemas
gtk3

  quickshell hyprland qt6-svg hyprlock python git \
  grim slurp wl-clipboard cliphist wf-recorder xdg-desktop-portal-hyprland python-dbus python-gobject \
  wireplumber pipewire libpulse brightnessctl networkmanager upower power-profiles-daemon \
  polkit hyprpolkitagent xdg-utils unzip pciutils gsettings-desktop-schemas gtk3 wireguard-tools zenity kdeconnect

# Lenh cai dat them nhom [optional] (Font + Btrfs + FFmpeg):
sudo pacman -S --needed --noconfirm \
  ffmpeg 7zip playerctl ttf-opensans ttf-jetbrains-mono btrfs-progs pacman-contrib libnotify
